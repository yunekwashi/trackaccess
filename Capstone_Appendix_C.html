Appendix C: Code Snippets

This section provides key representative code samples from the TrackAccess system, covering the hardware abstraction layer (Arduino), the RESTful backend endpoints (PHP), and the cross-platform application interface (Flutter).

Snippet C.1: Arduino RFID Hardware Firmware

Description: This C++ snippet runs on the Arduino microcontroller. It initializes the MFRC522 RFID module via SPI, constantly polls for a new physical card in the loop(), and transmits the raw Hexadecimal UID string over the USB Serial Port to the listening application machine securely. 

#include <MFRC522v2.h>
#include <MFRC522DriverSPI.h>
#include <MFRC522DriverPinSimple.h>

MFRC522DriverPinSimple ss_pin(5);
MFRC522DriverSPI driver{ss_pin}; 
MFRC522 mfrc522{driver};        

void setup() {
  Serial.begin(115200); 
  while (!Serial);
  mfrc522.PCD_Init();   
  Serial.println(F("TrackAccess RFID Scanner Ready"));
}

void loop() {
  if (!mfrc522.PICC_IsNewCardPresent() || !mfrc522.PICC_ReadCardSerial()) {
    return;
  }

  // Send UID to Flutter app: "XX XX XX XX"
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    if (i > 0) Serial.print(" ");
    if (mfrc522.uid.uidByte[i] < 0x10) Serial.print("0");
    Serial.print(mfrc522.uid.uidByte[i], HEX);
  }
  Serial.println(); // Newline triggers execution

  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();
  delay(1500); // Cooldown to block spam scanning
}

Snippet C.2: Process Scan & Serial Listener (Flutter/Dart)

Description: This snippet shows the Flutter application securely monitoring computer COM ports to capture the incoming RFID serial data. When a newline character is detected, it sanitizes the UID and triggers _processScan() to validate against the database and log attendance.

void _scanForScanner() {
  final ports = SerialPort.availablePorts;
  for (final name in ports) {
    final port = SerialPort(name);
    if (port.openReadWrite()) {
      // Configure Serial Connection 115200 Baud
      port.config.baudRate = 115200;
      
      final reader = SerialPortReader(port);
      _serialSub = reader.stream.listen((data) {
        final raw = utf8.decode(data);
        _serialBuffer.write(raw);
        
        // Execute block on physical tap (newline detection)
        if (raw.contains('\n')) {
          final lines = _serialBuffer.toString().split('\n');
          final uid = lines[lines.length - 2].trim();
          
          if (uid.isNotEmpty && uid.length >= 4) {
             _processScan(uid); // Authorize ID and log visit
          }
        }
      });
      return; 
    }
  }
}

Snippet C.3: API Endpoint Integration (PHP Backend/MySQL)

Description: A server-side RESTful PHP endpoint responsible for parsing raw physical scan injections. The script connects to the MySQL backend, sanitizes the payload to prevent SQL Injection, and inserts an immutable audit log reflecting real-world events.

<?php
header('Content-Type: application/json');
require_once 'db_config.php'; // Connection Wrapper

// Listen for POST payload containing the RFID string validation
$data = json_decode(file_get_contents("php://input"), true);

if ($data && isset($data["uid"])) {
    $uid = $conn->real_escape_string($data["uid"]);
    
    // Save structural record of physical event
    $sql = "INSERT INTO scans (uid, created_at) VALUES ('$uid', NOW())";
    
    if ($conn->query($sql) === TRUE) {
        echo json_encode(["success" => true, "message" => "Scan recorded securely."]);
    } else {
        http_response_code(500);
        echo json_encode(["success" => false, "message" => "Failed to write audit log."]);
    }
} else {
    http_response_code(400);
    echo json_encode(["success" => false, "message" => "Invalid parameters"]);
}
$conn->close();
?>
